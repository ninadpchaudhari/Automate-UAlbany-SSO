// ==UserScript==
// @name         UAlbany SSO Auto-Login (Delayed)
// @namespace    http://tampermonkey.net/
// @version      1.3
// @description  Waits 1 second after autofill before clicking sign in
// @author       Ninad
// @match        https://*.albany.edu/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // FLAG: Prevents the loop from setting 100 timers while we wait
    let isPending = false;

    function checkAndSubmit() {
        // --- STEP 1: The Landing Page (unchanged) ---
        const step1Button = document.evaluate(
            "//a[contains(text(), 'UAlbany Login')] | //button[contains(text(), 'UAlbany Login')]",
            document, null, XPathResult.FIRST_ORDERED_NODE_TYPE, null
        ).singleNodeValue;

        if (step1Button && !step1Button.dataset.clicked) {
            step1Button.dataset.clicked = "true";
            step1Button.click();
            return;
        }

        // --- STEP 2: The Login Form ---
        const passField = document.getElementById('inputPassword');
        const userField = document.getElementById('inputnetid');
        const submitBtn = document.querySelector('button[name="_eventId_proceed"]');

        if (passField && submitBtn && userField) {

            // Check if filled AND if we aren't already waiting to submit
            if (passField.value.length > 0 && userField.value.length > 0 && !isPending) {

                console.log("Credentials detected. Starting countdown...");
                isPending = true; // Lock the logic

                // Visual Feedback: Let you know it's happening
                submitBtn.innerText = "Auto-Submitting in 1s...";
                submitBtn.style.transition = "background-color 0.5s";
                submitBtn.style.backgroundColor = "#e0a800"; // Warning color (Yellow/Orange)

                // The Delay
                setTimeout(() => {
                    submitBtn.innerText = "Signing in...";
                    submitBtn.style.backgroundColor = "#28a745"; // Success color (Green)
                    submitBtn.click();
                }, 1000); // 1000ms = 1 second
            }
        }
    }

    setInterval(checkAndSubmit, 200);
})();
